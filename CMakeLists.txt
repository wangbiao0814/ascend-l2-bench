cmake_minimum_required(VERSION 3.20)
project(ascend_l2_bench LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()
if(NOT SOC_VERSION)
    message(FATAL_ERROR "Set SOC_VERSION to the exact target SoC supported by your CANN")
endif()
include(cmake/config_ascend.cmake)
if(NOT IS_DIRECTORY "${ASCEND_HOME_PATH}/include")
    message(FATAL_ERROR "Load your CANN set_env.sh first (ASCEND_HOME_PATH is required)")
endif()
set(PROJECT_OP_SRC_BASE ${PROJECT_SOURCE_DIR}/csrc)
add_subdirectory(csrc)
