cmake_minimum_required(VERSION 3.20)
project(ascend_l2_bench LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()
if(NOT SOC_VERSION)
    message(FATAL_ERROR "Set SOC_VERSION to the exact target SoC supported by your CANN")
endif()
set(PROJECT_OP_SRC_BASE ${PROJECT_SOURCE_DIR}/csrc)
include(cmake/config_envs.cmake)
include(cmake/config_ascend.cmake)
if(NOT ASCEND_INCLUDE_DIR)
    set(ASCEND_INCLUDE_DIR ${ASCEND_HOME_PATH}/include)
endif()
add_subdirectory(csrc)
